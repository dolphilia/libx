      - name: Setup pinned Pandoc for importer tests
        shell: bash
        run: |
          set -euo pipefail
          case "$(uname -m)" in
            arm64) architecture=arm64; checksum=3eaeb3bd10982aecba5dd76158745a4f805afb39ab72a519bba2533c98ce002d ;;
            x86_64) architecture=x86_64; checksum=922e35c0210d7ca20ee9327811361d6d7f0ef0adda089e0e74cb3756c3d713f9 ;;
            *) echo "Unsupported macOS architecture" >&2; exit 1 ;;
          esac
          pandoc_directory="$RUNNER_TEMP/pandoc-3.8.3"
          mkdir -p "$pandoc_directory"
          pandoc_archive="$pandoc_directory/pandoc.zip"
          curl --fail --location --proto '=https' --proto-redir '=https' \
            "https://github.com/jgm/pandoc/releases/download/3.8.3/pandoc-3.8.3-$architecture-macOS.zip" \
            --output "$pandoc_archive"
          printf '%s  %s\n' "$checksum" "$pandoc_archive" | shasum -a 256 --check
          unzip -q "$pandoc_archive" -d "$pandoc_directory"
          pandoc_binary="$(find "$pandoc_directory" -type f -path '*/bin/pandoc' -print)"
          test -x "$pandoc_binary"
          "$pandoc_binary" --version
          printf '%s\n' "$(dirname "$pandoc_binary")" >> "$GITHUB_PATH"
